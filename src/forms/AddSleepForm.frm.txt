Private Sub UserForm_Initialize()
    Dim day As Variant
    cbDayOfWeek.RowSource = vbNullString
    cbDayOfWeek.Clear
    For Each day In Array("Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday")
        cbDayOfWeek.AddItem day
    Next day
End Sub

Private Sub AddSleep_Click()
    Dim hours As Double
    On Error GoTo Failed
    If Not IsSleepHours(txtSleep.value, hours) Then MsgBox "Enter numeric sleep hours between 0 and 24.", vbExclamation: Exit Sub
    SaveSleep cbDayOfWeek.value, hours
    Me.Hide
    Exit Sub
Failed:
    MsgBox Err.Description, vbExclamation
End Sub
